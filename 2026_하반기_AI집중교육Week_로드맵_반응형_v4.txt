function openQuiz(){
  document.getElementById("quizModal").classList.add("show");
  restartQuiz();
}
function closeQuiz(){document.getElementById("quizModal").classList.remove("show")}
function renderQ(){
  const q=quiz[qIndex];
  document.getElementById("qno").textContent=`QUESTION ${qIndex+1} / ${quiz.length}`;
  document.getElementById("question").textContent=q.q;
  const box=document.getElementById("options");
  box.innerHTML="";
  q.opts.forEach((o,i)=>{
    const el=document.createElement("div");
    el.className="opt"+(answers[qIndex]===i?" selected":"");
    el.innerHTML=`<div class="ot">${o[0]}</div><div class="os">${o[1]}</div>`;
    el.onclick=()=>{answers[qIndex]=i;renderQ()};
    box.appendChild(el);
  });
  document.getElementById("prevBtn").style.visibility=qIndex===0?"hidden":"visible";
  document.getElementById("nextBtn").disabled=answers[qIndex]===null;
  document.getElementById("nextBtn").textContent=qIndex===quiz.length-1?"결과 보기":"다음";
  document.getElementById("progressBar").style.width=`${((qIndex+1)/quiz.length)*100}%`;
}
function nextQ(){
  if(answers[qIndex]===null)return;
  if(qIndex<quiz.length-1){qIndex++;renderQ()} else showResult();
}
function prevQ(){if(qIndex>0){qIndex--;renderQ()}}
function restartQuiz(){
  qIndex=0;
  answers=Array(quiz.length).fill(null);
  document.getElementById("quizArea").style.display="block";
  document.getElementById("resultArea").style.display="none";
  renderQ();
}
function showResult(){
  let score=0;
  answers.forEach((a,idx)=>score+=quiz[idx].opts[a][2]);
  if(score<=4) finalLevel="beginner";
  else if(score<=8) finalLevel="intermediate";
  else if(score<=12) finalLevel="advanced";
  else finalLevel="leader";

  const r=resultInfo[finalLevel];
  document.getElementById("quizArea").style.display="none";
  document.getElementById("resultArea").style.display="block";
  document.getElementById("resultBadge").textContent=r.badge;
  document.getElementById("resultTitle").textContent=r.title;
  document.getElementById("resultDesc").textContent=r.desc;
  document.getElementById("resultCourses").innerHTML=r.courses.map(c=>`
    <div class="rc">
      <div class="rank">${c[0]}</div>
      <div class="name">${c[1]}</div>
      <div class="why">${c[2]}</div>
    </div>`).join("");
}
function goToLevel(){
  closeQuiz();
  document.getElementById("matrixSection").scrollIntoView({behavior:"smooth",block:"start"});
  const targetLevel = finalLevel==="leader" ? "advanced" : finalLevel;
  setTimeout(()=>{
    const cells=document.querySelectorAll(`[data-level="${targetLevel}"]`);
    cells.forEach(el=>el.classList.add("row-highlight"));
    setTimeout(()=>cells.forEach(el=>el.classList.remove("row-highlight")),2600);
  },350);
}
